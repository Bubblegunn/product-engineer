from decimal import Decimal, ROUND_HALF_UP

from . import tax


def _cents(x):
    # round() on a float sees 1.005 as 1.00499999..., so it rounds down.
    # Going through repr() recovers the decimal the caller meant.
    return float(Decimal(repr(x)).quantize(Decimal("0.01"), rounding=ROUND_HALF_UP))


def line_total(line):
    return _cents(line.qty * line.unit_price)


def build(lines, region):
    subtotal = _cents(sum(line_total(l) for l in lines))
    vat = _cents(subtotal * tax.rate_for(region))
    return {"subtotal": subtotal, "tax": vat, "total": _cents(subtotal + vat)}
EOF

    cents = Decimal(str(amount)).quantize(Decimal("0.01"), rounding=ROUND_HALF_UP)
    return float(cents)


def line_total(line):
    return round_half_up(line.qty * line.unit_price)


def build(lines, region):
    subtotal = round(sum(line_total(l) for l in lines), 2)
    vat = round(subtotal * tax.rate_for(region), 2)
    return {"subtotal": subtotal, "tax": vat, "total": round(subtotal + vat, 2)}

# Rounding

Invoice line totals round half up to two decimals (1.005 becomes 1.01).
`round_half_up` in `billing/invoice.py` does this with `Decimal`.

The invoice subtotal, tax and total, refund amounts, and the CSV export still
use Python's `round(x, 2)`, which is banker's rounding. Half-cent values may
therefore go either way there.
The bank reconciliation is the exception; see reconciliation.md.
