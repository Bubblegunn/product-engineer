def column_widths(records, columns=None):
    cols = columns or columns_of(records)
    widths = {}
    for col in cols:
        longest = max([len(str(col))] + [len(str(r.get(col, ""))) for r in records])
        widths[col] = min(longest + 2, 60)
    return widths


def _column_letter(index):
    """Return the spreadsheet column name (A, B, ..., Z, AA, ...) for a 0-based index."""
    name = ""
    index += 1
    while index:
        index, rem = divmod(index - 1, 26)
        name = chr(65 + rem) + name
    return name


def _text(value):
    return escape(_ILLEGAL_XML.sub("", str(value)))


def _string_cell(ref, value, style=_STYLE_DEFAULT):
    return '<c r="%s" s="%d" t="inlineStr"><is><t xml:space="preserve">%s</t></is></c>' % (
        ref, style, _text(value))


def _number_cell(ref, number, style=_STYLE_DEFAULT):
    return '<c r="%s" s="%d"><v>%s</v></c>' % (ref, style, repr(number))


def _cell(ref, value):
    """Render one data cell; returns '' for None so the cell stays empty."""
    if value is None:
        return ""
    if isinstance(value, bool):
        return '<c r="%s" t="b"><v>%d</v></c>' % (ref, value)
    if isinstance(value, (int, float)):
        if isinstance(value, float) and not math.isfinite(value):
            return _string_cell(ref, value)
        return _number_cell(ref, value)
    if isinstance(value, datetime.datetime):
        if value.tzinfo is not None:
            value = value.astimezone(datetime.timezone.utc).replace(tzinfo=None)
        serial = (value - _EPOCH).total_seconds() / 86400.0
        return _number_cell(ref, serial, _STYLE_DATETIME)
    if isinstance(value, datetime.date):
        serial = (datetime.datetime(value.year, value.month, value.day) - _EPOCH).days
        return _number_cell(ref, serial, _STYLE_DATE)
    return _string_cell(ref, value)


def _sheet_xml(records, cols):
    widths = column_widths(records, cols)
    parts = ['<?xml version="1.0" encoding="UTF-8" standalone="yes"?>'
             '<worksheet xmlns="http://schemas.openxmlformats.org/spreadsheetml/2006/main">']
    if cols:
        parts.append("<cols>")
        for i, col in enumerate(cols, start=1):
            parts.append('<col min="%d" max="%d" width="%d" customWidth="1"/>' % (i, i, widths[col]))
        parts.append("</cols>")
    parts.append("<sheetData>")
    header = "".join(_string_cell("%s1" % _column_letter(i), col, _STYLE_HEADER)
                     for i, col in enumerate(cols))
    parts.append('<row r="1">%s</row>' % header)
    for row_no, rec in enumerate(records, start=2):
        cells = "".join(_cell("%s%d" % (_column_letter(i), row_no), rec.get(col))
                        for i, col in enumerate(cols))
        parts.append('<row r="%d">%s</row>' % (row_no, cells))
    parts.append("</sheetData></worksheet>")
    return "".join(parts)


def write_xlsx(records, fh, columns=None):
    """Write records as a single-sheet .xlsx workbook to the binary file object fh.

    The first row holds the (bold) column names; numbers, booleans and
    dates/datetimes are stored as typed cells, everything else as text.
    Returns the number of records written, like write_csv.
    """
    cols = columns or columns_of(records)
    with zipfile.ZipFile(fh, "w", zipfile.ZIP_DEFLATED) as zf:
        zf.writestr("[Content_Types].xml", _CONTENT_TYPES)
        zf.writestr("_rels/.rels", _ROOT_RELS)
        zf.writestr("xl/workbook.xml", _WORKBOOK)
        zf.writestr("xl/_rels/workbook.xml.rels", _WORKBOOK_RELS)
        zf.writestr("xl/styles.xml", _STYLES)
        zf.writestr("xl/worksheets/sheet1.xml", _sheet_xml(records, cols))
    return len(records)

def read_sheet(data):
    """Parse a written workbook into {cell_ref: (type, style, text)}."""
    with zipfile.ZipFile(io.BytesIO(data) if isinstance(data, bytes) else data) as zf:
        root = ET.fromstring(zf.read("xl/worksheets/sheet1.xml"))
    cells = {}
    for c in root.iter("{%s}c" % NS["m"]):
        v = c.find("m:v", NS)
        t = c.find("m:is/m:t", NS)
        text = v.text if v is not None else (t.text if t is not None else None)
        cells[c.get("r")] = (c.get("t"), c.get("s"), text)
    return cells


def write(records, **kw):
    buf = io.BytesIO()

def _serial(value):
    """Convert a date or datetime to an Excel serial number."""
    if not isinstance(value, datetime.datetime):
        value = datetime.datetime(value.year, value.month, value.day)
    delta = value - _EPOCH
    return delta.days + delta.seconds / 86400.0


def _cell(ref, value):
    """Render one cell as XML: numbers and dates get typed cells, everything else inline text."""
    if value is None:
        return ""
    if isinstance(value, bool):
        return '<c r="%s" t="b"><v>%d</v></c>' % (ref, value)
    if isinstance(value, int):
        return '<c r="%s" s="1"><v>%d</v></c>' % (ref, value) if abs(value) >= 1000 else '<c r="%s"><v>%d</v></c>' % (ref, value)
    if isinstance(value, float):
        return '<c r="%s" s="1"><v>%r</v></c>' % (ref, value)
    if isinstance(value, datetime.datetime):
        return '<c r="%s" s="3"><v>%r</v></c>' % (ref, _serial(value))
    if isinstance(value, datetime.date):
        return '<c r="%s" s="2"><v>%r</v></c>' % (ref, _serial(value))
    return '<c r="%s" t="inlineStr"><is><t xml:space="preserve">%s</t></is></c>' % (ref, escape(str(value)))

def write_xlsx(records, fh, columns=None):
    """Write records as a single-sheet .xlsx workbook to the binary file ``fh``.

    The first row holds the column names (bold). Numbers and booleans are
    stored natively, dates and datetimes as Excel serials with a date format
    (timezone info is dropped), everything else as text. Returns the number of
    records written, like ``write_csv``.
    """
    cols = columns or columns_of(records)
    widths = column_widths(records, cols)

    rows = []
    header = "".join(
        f'<c r="{_column_letter(i)}1" s="3" t="inlineStr"><is><t xml:space="preserve">{_text(c)}</t></is></c>'
        for i, c in enumerate(cols)
    )
    rows.append(f'<row r="1">{header}</row>')
    for n, rec in enumerate(records, start=2):
        cells = "".join(
            _cell(f"{_column_letter(i)}{n}", rec.get(c)) for i, c in enumerate(cols)
        )
        rows.append(f'<row r="{n}">{cells}</row>')

    col_xml = "".join(
        f'<col min="{i}" max="{i}" width="{widths[c]}" customWidth="1"/>'
        for i, c in enumerate(cols, start=1)
    )
    sheet = (
        _HEAD
        + f'<worksheet xmlns="{_NS}">'
        + (f"<cols>{col_xml}</cols>" if cols else "")
        + f'<sheetData>{"".join(rows)}</sheetData></worksheet>'
    )

    with zipfile.ZipFile(fh, "w", zipfile.ZIP_DEFLATED) as zf:
        zf.writestr("[Content_Types].xml", _CONTENT_TYPES)
        zf.writestr("_rels/.rels", _ROOT_RELS)
        zf.writestr("xl/workbook.xml", _WORKBOOK)
        zf.writestr("xl/_rels/workbook.xml.rels", _WORKBOOK_RELS)
        zf.writestr("xl/styles.xml", _STYLES)
        zf.writestr("xl/worksheets/sheet1.xml", sheet)
    return len(records)

def read_sheet(data):
    """Return rows as lists of (type, style, value) from the first sheet."""
    with zipfile.ZipFile(io.BytesIO(data)) as zf:
