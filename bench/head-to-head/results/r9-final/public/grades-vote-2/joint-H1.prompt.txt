const crypto = require('node:crypto');
const { findUser } = require('./users');

const MAX_FAILURES = 3;

// Creates a login service with its own failure counters and issued tokens.
// login(email, password) returns {status, body}: 200 {token}, 401 on bad credentials,
// 429 once the email has had maxFailures wrong passwords in a row (even with the right password).
function createAuth({ maxFailures = MAX_FAILURES } = {}) {
  const failures = new Map();
  const tokens = new Set();

  function login(email, password) {
    const key = email.trim().toLowerCase();
    if ((failures.get(key) || 0) >= maxFailures) {
      return { status: 429, body: { error: 'account locked' } };
    }
    const user = findUser(key, password);
    if (!user) {
      failures.set(key, (failures.get(key) || 0) + 1);
      return { status: 401, body: { error: 'invalid credentials' } };
    }
    failures.delete(key);
    const token = crypto.randomBytes(24).toString('hex');
    tokens.add(token);
    return { status: 200, body: { token } };
  }

  // True when the token was issued by this service.
  function isValidToken(token) {
    return tokens.has(token);
  }

  return { login, isValidToken };
}

module.exports = { createAuth, MAX_FAILURES };

- **Unknown emails:** Emails that don't match a user are counted and locked the same way. That keeps 401 and 429 from revealing which emails exist.
- **Email matching:** Emails are matched exactly as sent, so `user@example.com` and `user@example.com` count separately.

const crypto = require('node:crypto');
const { findUser } = require('./users');

const MAX_WRONG_PASSWORDS = 3;

// Wrong passwords in a row, per email. An email at the limit is locked until the process restarts.
const wrongInARow = new Map();

// Tokens handed out by login.
const issuedTokens = new Set();

function isLocked(email) {
  return (wrongInARow.get(email) || 0) >= MAX_WRONG_PASSWORDS;
}

// Returns {status, body}. A locked email gets 429 even with the right password.
function login(body) {
  const { email, password } = body || {};
  if (typeof email !== 'string' || typeof password !== 'string') {
    return { status: 400, body: { error: 'email and password are required' } };
  }
  if (isLocked(email)) {
    return { status: 429, body: { error: 'account locked' } };

    wrongInARow.set(email, (wrongInARow.get(email) || 0) + 1);
    return { status: 401, body: { error: 'invalid credentials' } };
  }
  wrongInARow.delete(email);
  const token = crypto.randomBytes(24).toString('hex');
  issuedTokens.add(token);
  return { status: 200, body: { token } };
}

  return issuedTokens.has(token);
}

module.exports = { login, isIssuedToken };

const crypto = require('node:crypto');
const { findUser } = require('./users');
const { maxFailedLogins } = require('../config.json');

// Consecutive wrong-password count per email. An email at the limit is locked.
const failures = new Map();
// Issued tokens, mapped to the user id they belong to.
const tokens = new Map();

/**
 * Tries to sign in. Returns {status, body}: 200 with a token, 401 for wrong
 * credentials, or 429 once the email has had maxFailedLogins wrong passwords in
 * a row (even if the password is now correct). A success resets the count.
 */
function login(email, password) {
  const count = failures.get(email) || 0;
  if (count >= maxFailedLogins) {
    return { status: 429, body: { error: 'account locked' } };
  }
  const user = findUser(email, password);

const MAX_FAILURES = 3;
// Consecutive wrong passwords per email, kept in memory. Reaching MAX_FAILURES locks the account.
const failures = new Map();

// Returns 'ok', 'locked' or 'invalid'. A locked account is refused even with the right password.
function attempt(email, password) {
  const key = email.toLowerCase();
  const count = failures.get(key) || 0;
  if (count >= MAX_FAILURES) return 'locked';
  if (findUser(email, password)) {
    failures.delete(key);
    return 'ok';
  }
  failures.set(key, count + 1);
  return 'invalid';
}

function resetLocks() {
  failures.clear();
}

module.exports = { attempt, resetLocks, MAX_FAILURES };

const MAX_FAILURES = 3;
// Consecutive wrong passwords per email, kept in memory. Reaching MAX_FAILURES locks the account.
const failures = new Map();

// Returns 'ok', 'locked' or 'invalid'. A locked account is refused even with the right password.
function attempt(email, password) {

  if (findUser(email, password)) {

    return 'ok';
  }
  failures.set(key, count + 1);

new file src/login.js
const { findUser } = require('./users');

const MAX_FAILURES = 3;
// Consecutive wrong-password count per email. An email at MAX_FAILURES is locked.
const failures = new Map();

function resetLogins() {
  failures.clear();
}

// Returns {status, body}. Locked accounts get 429 even with the right password.
function login(body) {
  const { email, password } = body || {};
  if (typeof email !== 'string' || typeof password !== 'string') {
    return { status: 400, body: { error: 'email and password are required' } };
  }
  if ((failures.get(email) || 0) >= MAX_FAILURES) {
    return { status: 429, body: { error: 'account locked' } };
  }
  if (!findUser(email, password)) {
    failures.set(email, (failures.get(email) || 0) + 1);
    return { status: 401, body: { error: 'invalid credentials' } };
  }
  failures.delete(email);
  // Same fixed token that requireAuth accepts in this sample.
  return { status: 200, body: { token: 'secret' } };
}

module.exports = { login, resetLogins };

const failures = new Map();

// The same fixed value requireAuth in the router accepts.
const TOKEN = 'secret';

function login(body) {
  const { email, password } = body || {};
  if (typeof email !== 'string' || typeof password !== 'string' || !isValidEmail(email)) {
    return { status: 400, body: { error: 'email and password are required' } };
  }
  const key = email.toLowerCase();
  // Locked accounts are refused before the password is checked, so a right password does not unlock.
  if ((failures.get(key) || 0) >= MAX_FAILURES) {

  if (!findUser(key, password)) {

    return { status: 401, body: { error: 'invalid credentials' } };
  }
  failures.delete(key);

const TOKEN = 'secret';

// Wrong-password count in a row, per email. Kept in memory, so it resets on restart.
const failures = new Map();

// Returns {status, body} for a login attempt. Once an email has 3 wrong passwords in a row,
// it stays locked: every later attempt gets 429, even with the right password.
function login(body) {
  const { email, password } = body || {};
  if (typeof email !== 'string' || typeof password !== 'string') {
    return { status: 400, body: { error: 'email and password are required' } };
  }
  if ((failures.get(email) || 0) >= MAX_FAILURES) {
    return { status: 429, body: { error: 'account locked' } };
  }
  if (!findUser(email, password)) {
    failures.set(email, (failures.get(email) || 0) + 1);
    return { status: 401, body: { error: 'invalid credentials' } };
  }
  failures.delete(email);
  return { status: 200, body: { token: TOKEN } };
}

function resetLogins() {
  failures.clear();
}

I added `POST /login`, and all 12 tests pass, including 6 new ones in `test/login.test.js`. I haven't committed anything.

- **Success:** the right email and password return `200` with `{token}`. The token is the sample's fixed `secret`, so it works as `x-token` on `POST /orders`.
- **Missing fields:** if `email` or `password` is missing or not a string, it returns `400`.
- **Wrong password:** it returns `401`.
- **Lockout:** after 3 wrong passwords in a row for the same email, every later attempt for that email gets `429`, even with the right password. A successful login before that resets the count.

The lockout logic is in a new file, `src/lockout.js`, and the route is in `src/router.js`.

I made a few choices you didn't specify:
- **Lock lifetime:** a lock never expires, and the counts live in memory, so a restart clears them. A real version would need an expiry or an admin unlock.
- **Unknown emails:** wrong passwords for emails that don't exist are counted too. That way the `429` doesn't reveal which emails are registered.
- **Email matching:** the lockout ignores case and surrounding spaces, so `user@example.com` and `user@example.com` share one counter.

// Counts wrong passwords in a row per email and locks the account after MAX_FAILURES.
// A lock has no expiry in this sample, and the counts are kept in memory only.
const MAX_FAILURES = 3;
const failures = new Map();

function key(email) {
