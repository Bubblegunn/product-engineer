+'use strict';
+const config = require('../config');
+
+const hits = new Map();
+
+function clientIp(req) {
+  const xff = req.headers && req.headers['x-forwarded-for'];
+  if (config.trustProxy && xff) return xff.split(',')[0].trim();
+  return (req.socket && req.socket.remoteAddress) || 'unknown';
+}
+
+function rateLimit(req, res, next) {
+  const now = Date.now();
+  const ip = clientIp(req);
+  let entry = hits.get(ip);
+  if (!entry || now - entry.start >= config.rateLimit.windowMs) {
+    entry = { start: now, count: 0 };
+    hits.set(ip, entry);
+  }
+  entry.count += 1;
+  if (entry.count > config.rateLimit.max) {
+    const retry = Math.ceil((entry.start + config.rateLimit.windowMs - now) / 1000);
+    res.statusCode = 429;
+    res.setHeader('Retry-After', String(retry));
+    res.end(JSON.stringify({ error: 'rate_limited' }));
+    return;
+  }
+  next();
+}
+
+function reset() {
+  hits.clear();
+}
+
+module.exports = { rateLimit, reset };

+function clientIp(req) {

+  return (req.socket && req.socket.remoteAddress) || 'unknown';

+function rateLimit(req, res, next) {
+  const now = Date.now();
+  const ip = clientIp(req);
+  let entry = hits.get(ip);
+  if (!entry || now - entry.start >= config.rateLimit.windowMs) {
+    entry = { start: now, count: 0 };
+    hits.set(ip, entry);

+    res.statusCode = 429;
+    res.setHeader('Retry-After', String(retry));
+    res.end(JSON.stringify({ error: 'rate_limited' }));
+    return;
+  }
+  next();
+}

+  hits.clear();
+}

+const hits = new Map();

+function clientIp(req) {
+  const xff = req.headers && req.headers['x-forwarded-for'];
+  if (config.trustProxy && xff) return xff.split(',')[0].trim();
+  return (req.socket && req.socket.remoteAddress) || 'unknown';
+}
+
+function rateLimit(req, res, next) {
+  const now = Date.now();
+  const ip = clientIp(req);
+  let entry = hits.get(ip);
+  if (!entry || now - entry.start >= config.rateLimit.windowMs) {
+    entry = { start: now, count: 0 };
+    hits.set(ip, entry);
+  }
+  entry.count += 1;
+  if (entry.count > config.rateLimit.max) {
+    const retry = Math.ceil((entry.start + config.rateLimit.windowMs - now) / 1000);
+    res.statusCode = 429;
+    res.setHeader('Retry-After', String(retry));
+    res.end(JSON.stringify({ error: 'rate_limited' }));
+    return;
+  }
+  next();
+}

+  hits.clear();
+}

+'use strict';
+const config = require('../config');
+
+const hits = new Map();
+
+function clientIp(req) {
+  const xff = req.headers && req.headers['x-forwarded-for'];
+  if (config.trustProxy && xff) return xff.split(',')[0].trim();
+  return (req.socket && req.socket.remoteAddress) || 'unknown';
+}
+
+function rateLimit(req, res, next) {
+  const now = Date.now();
+  const ip = clientIp(req);
+  let entry = hits.get(ip);
+  if (!entry || now - entry.start >= config.rateLimit.windowMs) {
+    entry = { start: now, count: 0 };
+    hits.set(ip, entry);
+  }
+  entry.count += 1;
+  if (entry.count > config.rateLimit.max) {
+    const retry = Math.ceil((entry.start + config.rateLimit.windowMs - now) / 1000);
+    res.statusCode = 429;
+    res.setHeader('Retry-After', String(retry));
+    res.end(JSON.stringify({ error: 'rate_limited' }));
+    return;
+  }
+  next();
+}
+
+function reset() {
+  hits.clear();
+}
+
+module.exports = { rateLimit, reset };

+'use strict';
+const config = require('../config');
+
+const hits = new Map();
+
+function clientIp(req) {
+  const xff = req.headers && req.headers['x-forwarded-for'];
+  if (config.proxy.trust && xff) return xff.split(',')[0].trim();
+  return (req.socket && req.socket.remoteAddress) || 'unknown';
+}
+
+function rateLimit(req, res, next) {
+  const now = Date.now();
+  const ip = clientIp(req);
+  let entry = hits.get(ip);
+  if (!entry || now - entry.start >= config.rateLimit.windowMs) {
+    entry = { start: now, count: 0 };
+    hits.set(ip, entry);
+  }
+  entry.count += 1;
+  if (entry.count > config.rateLimit.max) {
+    const retry = Math.ceil((entry.start + config.rateLimit.windowMs - now) / 1000);
+    res.statusCode = 429;
+    res.setHeader('Retry-After', String(retry));
+    res.end(JSON.stringify({ error: 'rate_limited' }));
+    return;
+  }
+  next();
+}
+
+function reset() {
+  hits.clear();
+}
+
+module.exports = { rateLimit, reset };
