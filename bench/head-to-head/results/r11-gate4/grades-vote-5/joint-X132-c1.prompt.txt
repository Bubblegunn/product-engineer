+'use strict';
+const config = require('../config');
+
+const hits = new Map();
+
+function clientIp(req) {
+  const xff = req.headers && req.headers['x-forwarded-for'];
+  if (config.trustProxy && xff) return xff.split(',')[0].trim();
+  return (req.socket && req.socket.remoteAddress) || 'unknown';
+}
+
+function rateLimit(req, res, next) {
+  const now = Date.now();
+  const ip = clientIp(req);
+  let entry = hits.get(ip);
+  if (!entry || now - entry.start >= config.rateLimit.windowMs) {
+    entry = { start: now, count: 0 };
+    hits.set(ip, entry);
+  }
+  entry.count += 1;
+  if (entry.count > config.rateLimit.max) {
+    const retry = Math.ceil((entry.start + config.rateLimit.windowMs - now) / 1000);
+    res.statusCode = 429;
+    res.setHeader('Retry-After', String(retry));
+    res.end(JSON.stringify({ error: 'rate_limited' }));
+    return;
+  }
+  next();
+}
+
+function reset() {
+  hits.clear();
+}
+
+module.exports = { rateLimit, reset };

+const config = require('../config');

+const hits = new Map();
+
+function clientIp(req) {
+  const xff = req.headers && req.headers['x-forwarded-for'];
+  if (config.trustProxy && xff) return xff.split(',')[0].trim();
+  return (req.socket && req.socket.remoteAddress) || 'unknown';
+}
+
+function rateLimit(req, res, next) {
+  const now = Date.now();
+  const ip = clientIp(req);
+  let entry = hits.get(ip);
+  if (!entry || now - entry.start >= config.rateLimit.windowMs) {
+    entry = { start: now, count: 0 };
+    hits.set(ip, entry);
+  }
+  entry.count += 1;
+  if (entry.count > config.rateLimit.max) {
+    const retry = Math.ceil((entry.start + config.rateLimit.windowMs - now) / 1000);
+    res.statusCode = 429;
+    res.setHeader('Retry-After', String(retry));
+    res.end(JSON.stringify({ error: 'rate_limited' }));
+    return;
+  }
+  next();
+}
+
+function reset() {
+  hits.clear();
+}

+'use strict';
+const config = require('../config');
+
+const hits = new Map();
+
+function clientIp(req) {
+  const xff = req.headers && req.headers['x-forwarded-for'];
+  if (config.trustProxy && xff) return xff.split(',')[0].trim();
+  return (req.socket && req.socket.remoteAddress) || 'unknown';
+}
+
+function rateLimit(req, res, next) {
+  const now = Date.now();
+  const ip = clientIp(req);
+  let entry = hits.get(ip);
+  if (!entry || now - entry.start >= config.rateLimit.windowMs) {
+    entry = { start: now, count: 0 };
+    hits.set(ip, entry);
+  }
+  entry.count += 1;
+  if (entry.count > config.rateLimit.max) {
+    const retry = Math.ceil((entry.start + config.rateLimit.windowMs - now) / 1000);
+    res.statusCode = 429;
+    res.setHeader('Retry-After', String(retry));
+    res.end(JSON.stringify({ error: 'rate_limited' }));
+    return;
+  }
+  next();
+}
+
+function reset() {
+  hits.clear();
+}
+
+module.exports = { rateLimit, reset };

+const config = require('../config');
+
+const hits = new Map();
+
+function clientIp(req) {
+  const xff = req.headers && req.headers['x-forwarded-for'];
+  if (config.trustProxy && xff) return xff.split(',')[0].trim();
+  return (req.socket && req.socket.remoteAddress) || 'unknown';
+}
+
+function rateLimit(req, res, next) {
+  const now = Date.now();
+  const ip = clientIp(req);
+  let entry = hits.get(ip);
+  if (!entry || now - entry.start >= config.rateLimit.windowMs) {
+    entry = { start: now, count: 0 };
+    hits.set(ip, entry);
+  }
+  entry.count += 1;
+  if (entry.count > config.rateLimit.max) {
+    const retry = Math.ceil((entry.start + config.rateLimit.windowMs - now) / 1000);
+    res.statusCode = 429;
+    res.setHeader('Retry-After', String(retry));
+    res.end(JSON.stringify({ error: 'rate_limited' }));
+    return;
+  }
+  next();
+}
+
+function reset() {
+  hits.clear();
+}
+
+module.exports = { rateLimit, reset };

+const config = require('../config');

+const hits = new Map();

+function clientIp(req) {
+  const xff = req.headers && req.headers['x-forwarded-for'];
+  if (config.trustProxy && xff) return xff.split(',')[0].trim();
+  return (req.socket && req.socket.remoteAddress) || 'unknown';
+}
+
+function rateLimit(req, res, next) {
+  const now = Date.now();
+  const ip = clientIp(req);
+  let entry = hits.get(ip);
+  if (!entry || now - entry.start >= config.rateLimit.windowMs) {
+    entry = { start: now, count: 0 };
+    hits.set(ip, entry);
+  }
+  entry.count += 1;
+  if (entry.count > config.rateLimit.max) {
+    const retry = Math.ceil((entry.start + config.rateLimit.windowMs - now) / 1000);
+    res.statusCode = 429;
+    res.setHeader('Retry-After', String(retry));
+    res.end(JSON.stringify({ error: 'rate_limited' }));
+    return;
+  }
+  next();
+}
+
+function reset() {
+  hits.clear();
+}

+module.exports = { rateLimit, reset };

+'use strict';
+const config = require('../config');
+
+const hits = new Map();
+
+function clientIp(req) {
+  const xff = req.headers && req.headers['x-forwarded-for'];
+  if (config.proxy.trust && xff) return xff.split(',')[0].trim();
+  return (req.socket && req.socket.remoteAddress) || 'unknown';
+}
+
+function rateLimit(req, res, next) {
+  const now = Date.now();
+  const ip = clientIp(req);
+  let entry = hits.get(ip);
+  if (!entry || now - entry.start >= config.rateLimit.windowMs) {
+    entry = { start: now, count: 0 };
+    hits.set(ip, entry);
+  }
+  entry.count += 1;
+  if (entry.count > config.rateLimit.max) {
+    const retry = Math.ceil((entry.start + config.rateLimit.windowMs - now) / 1000);
+    res.statusCode = 429;
+    res.setHeader('Retry-After', String(retry));
+    res.end(JSON.stringify({ error: 'rate_limited' }));
+    return;
+  }
+  next();
+}
+
+function reset() {
+  hits.clear();
+}
+
+module.exports = { rateLimit, reset };

+'use strict';
+const test = require('node:test');
+const assert = require('node:assert');
+const config = require('../src/config');
+const { rateLimit, reset } = require('../src/middleware/ratelimit');
+
+function fakeRes() {
+  return {
+    statusCode: 200,
+    headers: {},
+    body: null,
+    setHeader(k, v) { this.headers[k] = v; },
+    end(b) { this.body = b; },
+  };
+}
+
+function req(addr, xff) {
+  return { headers: xff ? { 'x-forwarded-for': xff } : {}, socket: { remoteAddress: addr } };
+}
+
+test.beforeEach(() => {
+  reset();
+  config.rateLimit.max = 2;
+  config.proxy.trust = false;
+});
+
+test('allows requests up to the limit then answers 429', () => {
+  let passed = 0;
+  const last = fakeRes();
+  rateLimit(req('10.0.0.1'), fakeRes(), () => passed++);
+  rateLimit(req('10.0.0.1'), fakeRes(), () => passed++);
+  rateLimit(req('10.0.0.1'), last, () => passed++);
+  assert.strictEqual(passed, 2);
+  assert.strictEqual(last.statusCode, 429);
+  assert.ok(Number(last.headers['Retry-After']) >= 1);
+});
+
+test('separate clients behind a trusted proxy get separate buckets', () => {
+  config.proxy.trust = true;
+  config.rateLimit.max = 1;
+  let passed = 0;
+  rateLimit(req('10.0.0.9', '1.1.1.1'), fakeRes(), () => passed++);
+  rateLimit(req('10.0.0.9', '2.2.2.2'), fakeRes(), () => passed++);
+  assert.strictEqual(passed, 2);
+});
